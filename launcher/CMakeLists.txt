cmake_minimum_required(VERSION 3.28)
project(pt_launcher LANGUAGES CXX)
set(PT_SUPPORTED_LINUX_SHA256 "" CACHE STRING "Approved matching Linux runtime SHA-256")
set(PT_SUPPORTED_WINDOWS_SHA256 "" CACHE STRING "Approved matching Windows runtime SHA-256")
foreach(hash IN ITEMS PT_SUPPORTED_LINUX_SHA256 PT_SUPPORTED_WINDOWS_SHA256)
  string(LENGTH "${${hash}}" hash_length)
  if(NOT "${${hash}}" STREQUAL "" AND
     (NOT hash_length EQUAL 64 OR NOT "${${hash}}" MATCHES "^[0-9a-f]+$"))
    message(FATAL_ERROR "${hash} must be an empty string or a lowercase SHA-256")
  endif()
endforeach()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PT_LAUNCHER_QT_MIN_VERSION "6.12.0" CACHE STRING "Required Qt version for release builds")
find_package(Qt6 ${PT_LAUNCHER_QT_MIN_VERSION} REQUIRED COMPONENTS Widgets)
find_package(ZLIB REQUIRED)
add_library(pt_launcher_inspection STATIC inspection.cpp)
target_include_directories(pt_launcher_inspection PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../src ${CMAKE_CURRENT_SOURCE_DIR}/../installer/Native)
target_link_libraries(pt_launcher_inspection PUBLIC Qt6::Widgets PRIVATE ZLIB::ZLIB)
add_executable(pt_launcher main.cpp)
file(COPY ${CMAKE_CURRENT_SOURCE_DIR}/assets DESTINATION ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(pt_launcher PRIVATE pt_launcher_inspection)
target_compile_definitions(pt_launcher PRIVATE PT_SUPPORTED_LINUX_SHA256="${PT_SUPPORTED_LINUX_SHA256}" PT_SUPPORTED_WINDOWS_SHA256="${PT_SUPPORTED_WINDOWS_SHA256}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/supported-runtime-sha256.txt"
  "linux=${PT_SUPPORTED_LINUX_SHA256}\nwindows=${PT_SUPPORTED_WINDOWS_SHA256}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/supported-runtime-sha256.txt" DESTINATION .)
if(WIN32)
  set_target_properties(pt_launcher PROPERTIES OUTPUT_NAME pt-launcher)
  install(TARGETS pt_launcher RUNTIME DESTINATION .)
  install(DIRECTORY assets DESTINATION .)
else()
  set_target_properties(pt_launcher PROPERTIES OUTPUT_NAME pt-launcher INSTALL_RPATH "$ORIGIN/../lib")
  install(TARGETS pt_launcher RUNTIME DESTINATION libexec)
  install(DIRECTORY assets DESTINATION libexec)
endif()
add_executable(pt_launcher_inspection_test tests/inspection_test.cpp)
target_link_libraries(pt_launcher_inspection_test PRIVATE pt_launcher_inspection)
enable_testing()
add_test(NAME launcher_inspection COMMAND pt_launcher_inspection_test)
add_library(pt_launcher_session STATIC session.cpp)
target_link_libraries(pt_launcher_session PUBLIC pt_launcher_inspection)
target_include_directories(pt_launcher_session PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(WIN32)
  target_link_libraries(pt_launcher_session PRIVATE user32)
endif()
target_link_libraries(pt_launcher PRIVATE pt_launcher_session)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(pt_launcher_session_test tests/session_test.cpp)
  target_link_libraries(pt_launcher_session_test PRIVATE pt_launcher_session)
  add_executable(pt_launcher_ui_test_app main.cpp)
  target_link_libraries(pt_launcher_ui_test_app PRIVATE pt_launcher_session)
  target_compile_definitions(pt_launcher_ui_test_app PRIVATE PT_LAUNCHER_TEST_BUILD PT_SUPPORTED_LINUX_SHA256="" PT_SUPPORTED_WINDOWS_SHA256="")
  add_test(NAME launcher_session COMMAND pt_launcher_session_test $<TARGET_FILE:pt_launcher_ui_test_app>)
  add_executable(pt_launcher_game_window_fixture tests/game_window_fixture.cpp)
  target_link_libraries(pt_launcher_game_window_fixture PRIVATE Qt6::Widgets)
  add_executable(pt_launcher_window_test tests/window_test.cpp)
  target_link_libraries(pt_launcher_window_test PRIVATE pt_launcher_session)
  add_test(NAME launcher_window COMMAND ${CMAKE_CURRENT_SOURCE_DIR}/run-window-test.sh
    $<TARGET_FILE:pt_launcher_window_test> $<TARGET_FILE:pt_launcher_game_window_fixture>)
endif()
