cmake_minimum_required(VERSION 3.28)
project(pt_launcher LANGUAGES CXX)
set(PT_SUPPORTED_LINUX_SHA256 "" CACHE STRING "Approved matching Linux runtime SHA-256")
set(PT_SUPPORTED_WINDOWS_SHA256 "" CACHE STRING "Approved matching Windows runtime SHA-256")
foreach(hash IN ITEMS PT_SUPPORTED_LINUX_SHA256 PT_SUPPORTED_WINDOWS_SHA256)
  string(LENGTH "${${hash}}" hash_length)
  if(NOT "${${hash}}" STREQUAL "" AND
     (NOT hash_length EQUAL 64 OR NOT "${${hash}}" MATCHES "^[0-9a-f]+$"))
    message(FATAL_ERROR "${hash} must be an empty string or a lowercase SHA-256")
  endif()
endforeach()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PT_LAUNCHER_QT_MIN_VERSION "6.12.0" CACHE STRING "Required Qt version for release builds")
find_package(Qt6 ${PT_LAUNCHER_QT_MIN_VERSION} REQUIRED COMPONENTS Widgets)
find_package(ZLIB REQUIRED)
include(FetchContent)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_HIDAPI ON CACHE BOOL "" FORCE)
set(SDL_VIRTUAL_JOYSTICK ON CACHE BOOL "" FORCE)
set(SDL_HIDAPI_LIBUSB OFF CACHE BOOL "" FORCE)
FetchContent_Declare(sdl3 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG release-3.4.16 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(sdl3)
add_library(pt_launcher_inspection STATIC inspection.cpp)
target_include_directories(pt_launcher_inspection PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../src ${CMAKE_CURRENT_SOURCE_DIR}/../installer/Native)
target_link_libraries(pt_launcher_inspection PUBLIC Qt6::Widgets PRIVATE ZLIB::ZLIB)
add_executable(pt_launcher main.cpp)
file(COPY ${CMAKE_CURRENT_SOURCE_DIR}/assets DESTINATION ${CMAKE_CURRENT_BINARY_DIR})
add_library(pt_launcher_controller STATIC controller.cpp)
target_include_directories(pt_launcher_controller PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(pt_launcher_controller PUBLIC Qt6::Widgets SDL3::SDL3-static)
target_link_libraries(pt_launcher PRIVATE pt_launcher_inspection pt_launcher_controller)
target_compile_definitions(pt_launcher PRIVATE PT_SUPPORTED_LINUX_SHA256="${PT_SUPPORTED_LINUX_SHA256}" PT_SUPPORTED_WINDOWS_SHA256="${PT_SUPPORTED_WINDOWS_SHA256}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/supported-runtime-sha256.txt"
  "linux=${PT_SUPPORTED_LINUX_SHA256}\nwindows=${PT_SUPPORTED_WINDOWS_SHA256}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/supported-runtime-sha256.txt" DESTINATION .)
install(FILES "${sdl3_SOURCE_DIR}/LICENSE.txt" DESTINATION notices RENAME SDL3-LICENSE.txt)
if(WIN32)
  set_target_properties(pt_launcher PROPERTIES OUTPUT_NAME pt-launcher)
  install(TARGETS pt_launcher RUNTIME DESTINATION .)
  install(DIRECTORY assets DESTINATION .)
else()
  set_target_properties(pt_launcher PROPERTIES OUTPUT_NAME pt-launcher INSTALL_RPATH "$ORIGIN/../lib")
  install(TARGETS pt_launcher RUNTIME DESTINATION libexec)
  install(DIRECTORY assets DESTINATION libexec)
endif()

# The fixture programs use assert() for checks and operations such as waiting
# for processes. Keep those checks active when the shipped launcher is Release.
foreach(test IN ITEMS pt_launcher_inspection_test pt_launcher_session_test pt_launcher_window_test pt_launcher_settings_test pt_launcher_settings_dialog_test pt_launcher_controller_test)
  if(TARGET ${test})
    if(MSVC)
      target_compile_options(${test} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${test} PRIVATE -UNDEBUG)
    endif()
  endif()
endforeach()
